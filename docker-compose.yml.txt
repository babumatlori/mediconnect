services:

  mysql:
    image: mysql:8.0
    container_name: mediconnect-mysql
    restart: unless-stopped
    environment:
      MYSQL_ROOT_PASSWORD: mediconnect123
      MYSQL_DATABASE: mediconnect
    ports:
      - "3306:3306"
    volumes:
      - mysql_data:/var/lib/mysql
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 10s
      retries: 5

  redis:
    image: redis:7-alpine
    container_name: mediconnect-redis
    restart: unless-stopped
    ports:
      - "6379:6379"
    command: redis-server --appendonly yes
    volumes:
      - redis_data:/data

volumes:
  mysql_data:
